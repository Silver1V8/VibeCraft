def main() -> int:
    sandbox = pathlib.Path(sys.argv[1])
    revert = "--revert" in sys.argv
    target = sandbox / "rag" / "core" / "retrieval.py"
    backup = sandbox / "rag" / "core" / "retrieval.py.w1orig"
    if not target.is_file():
        print(f"ERRORE: {target} assente")
        return 2

    if revert:
        if not backup.is_file():
            print("nessun backup: gia' originale")
            return 0
        atteso = hashlib.sha256(backup.read_bytes()).hexdigest()
        target.write_bytes(backup.read_bytes())
        ok = hashlib.sha256(target.read_bytes()).hexdigest() == atteso
        backup.unlink()
        print(f"RIPRISTINATO byte-esatto: {ok} sha256={atteso[:16]}")
        return 0 if ok else 1

    testo = target.read_text(encoding="utf-8")
    if all(n in testo for _, _, n in SOSTITUZIONI):
        print("gia' applicata")
        return 0
    for desc, vecchio, _ in SOSTITUZIONI:
        if testo.count(vecchio) != 1:
            print(f"ERRORE: «{desc}» trovato {testo.count(vecchio)} volte, non 1")
            return 2
    if not backup.is_file():
        backup.write_bytes(target.read_bytes())
    for desc, vecchio, nuovo in SOSTITUZIONI:
        testo = testo.replace(vecchio, nuovo)
    target.write_text(testo, encoding="utf-8")
    print(f"APPLICATA W1 a {target}  ({len(SOSTITUZIONI)} sostituzioni)")
    print(f"  sha256 prima={hashlib.sha256(backup.read_bytes()).hexdigest()[:16]}")
    print(f"  sha256 dopo ={hashlib.sha256(target.read_bytes()).hexdigest()[:16]}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

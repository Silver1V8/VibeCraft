"""FASE 2/3 — ipotesi di SERIALIZZAZIONE (in perimetro): la posizione dei blocchi.

Il contesto oggi e': [chunk documentali] + [grafo] + «=== CODICE ESISTENTE ===».
La domanda sta in fondo al prompt. I 10 blocchi di codice separano quindi i
documenti dalla domanda. Spostare il codice PRIMA dei documenti non cambia il
ranking ne' quali fonti entrano: cambia solo l'ordine di serializzazione.
"""
import json
import pathlib
import sys
import re
RAG = pathlib.Path(r"C:\Users\Silve\Desktop\vibecraft\rag")
sys.path.insert(0, str(RAG))
from core import llm, retrieval  # noqa: E402
ARCH = RAG / "eval/results/fix_attribuzione_20260909-183000/out"
cfg, n, uscita = sys.argv[1], int(sys.argv[2]), sys.argv[3]
d = json.loads((ARCH/f"eval_{cfg}_p1.json").read_text(encoding="utf-8"))
q = d["domande"]["pdf-diag-3"]
ctx = q["contesto_inviato"]
MARK = "\n\n=== CODICE ESISTENTE (implementazione) ===\n\n"
i = ctx.find(MARK)
assert i > 0, "blocco codice non trovato"
docs, code = ctx[:i], ctx[i:]
codice_prima = MARK.strip("\n") + "\n\n" + code[len(MARK):] + "\n\n" + docs
scala=[("NORMS",0.9),("REQUIREMENTS",0.87),("METADATA",0.6),("METHODOLOGY",0.5),("SENZA CLASSE",0.3)]
COND = [("controllo: documenti, poi codice", ctx),
        ("E1: codice prima, documenti vicini alla domanda", codice_prima)]
res={}
for nome, c in COND:
    righe=[]
    for k in range(n):
        p = retrieval.build_ask_prompt(c, q["domanda"], authority_classes=scala)
        r = llm.generate(p, purpose="ask")
        kb="19.5 KB" in r
        coeff=bool(re.search(r"\b0[.,](?:30|50|60|85|87|90)\b", r))
        nega=bool(re.search(r"non\s+(?:è|e')?\s*(?:riport\w+|contien\w+|specific\w+|indic\w+|present\w+)|nessun\w*\s+valore", r, re.I))
        righe.append({"r":k+1,"kb":kb,"coeff":coeff,"nega":nega,"risposta":r})
    pulite=sum(x["kb"] and not x["coeff"] and not x["nega"] for x in righe)
    print(f"  {nome:46} 19.5KB {sum(x['kb'] for x in righe)}/{n} | coeff {sum(x['coeff'] for x in righe)}/{n} | PULITE {pulite}/{n}")
    res[nome]={"righe":righe,"contesto_char":len(c)}
pathlib.Path(uscita).write_text(json.dumps(res,indent=2,ensure_ascii=False),encoding="utf-8")

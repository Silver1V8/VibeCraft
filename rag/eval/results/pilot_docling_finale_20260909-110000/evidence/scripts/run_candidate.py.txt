"""Candidato finale: StandardPdfPipeline + stage NATIVO heading_hierarchy.

Differenze dichiarate rispetto alla configurazione congelata in
pilot_docling_standard_20260909-095127/run_standard.py:
  1. input = copie con template corretto (pilot_docling_sourcefix_20260909), hash distinti;
  2. heading_hierarchy_options.enabled = True  (default della libreria: False);
  3. generate_parsed_pages = True             (richiesto da use_style).
Nient'altro cambia: stessi pesi locali, stesso backend, stesse opzioni tabellari.

Memoria: campionatore continuo su thread separato, intervallo dichiarato nel report.
Non e' un massimo a pochi checkpoint.

Uso: run_candidate.py <out_dir> <artifacts_dir> <pdf_dir>
"""

import hashlib  # noqa: E402
import importlib.metadata  # noqa: E402
import json  # noqa: E402
import os  # noqa: E402
import pathlib  # noqa: E402
import platform  # noqa: E402
import sys  # noqa: E402
import threading  # noqa: E402
import time  # noqa: E402

import psutil  # noqa: E402

OUT = pathlib.Path(sys.argv[1]).resolve()
ARTIFACTS = pathlib.Path(sys.argv[2]).resolve()
PDFS = pathlib.Path(sys.argv[3]).resolve()
DOCS = {
    "diagrammi": PDFS / "catalogo_diagrammi_normativi.pdf",
    "mockup": PDFS / "catalogo_mockup_normativi.pdf",
}

from docling.backend.docling_parse_backend import ThreadedDoclingParseDocumentBackend  # noqa: E402
from docling.datamodel.accelerator_options import AcceleratorDevice, AcceleratorOptions  # noqa: E402
from docling.datamodel.base_models import InputFormat  # noqa: E402
from docling.datamodel.pipeline_options import (  # noqa: E402
    HeadingHierarchyOptions,
    PdfPipelineOptions,
    TableFormerMode,
    TableStructureOptions,
)
from docling.document_converter import DocumentConverter, PdfFormatOption  # noqa: E402
from docling.pipeline.standard_pdf_pipeline import StandardPdfPipeline  # noqa: E402
from docling_core.types.doc.document import SectionHeaderItem, TitleItem  # noqa: E402

OUT.mkdir(parents=True, exist_ok=True)
proc = psutil.Process()

SAMPLE_INTERVAL_S = 0.05
_samples: list[int] = []
_stop = threading.Event()


def _sampler():
    while not _stop.is_set():
        try:
            _samples.append(proc.memory_info().rss)
        except Exception:
            pass
        _stop.wait(SAMPLE_INTERVAL_S)


sampler = threading.Thread(target=_sampler, daemon=True)
sampler.start()

options = PdfPipelineOptions(
    artifacts_path=ARTIFACTS,
    do_ocr=False,
    do_table_structure=True,
    force_backend_text=True,
    table_structure_options=TableStructureOptions(mode=TableFormerMode.ACCURATE, do_cell_matching=True),
    accelerator_options=AcceleratorOptions(device=AcceleratorDevice.CPU),
    do_code_enrichment=False,
    do_formula_enrichment=False,
    do_picture_description=False,
    do_picture_classification=False,
    enable_remote_services=False,
    # --- le due sole differenze di configurazione, dichiarate ---
    generate_parsed_pages=True,
    heading_hierarchy_options=HeadingHierarchyOptions(enabled=True),
)
fmt = PdfFormatOption(
    pipeline_cls=StandardPdfPipeline,
    pipeline_options=options,
    backend=ThreadedDoclingParseDocumentBackend,
)

hh = options.heading_hierarchy_options
report = {
    "variante": "StandardPdfPipeline + heading_hierarchy nativo (enabled=True)",
    "python": platform.python_version(),
    "packages": {
        n: importlib.metadata.version(n)
        for n in ("docling", "docling-core", "docling-ibm-models", "docling-parse", "onnxruntime", "torch")
    },
    "pipeline": f"{StandardPdfPipeline.__module__}.{StandardPdfPipeline.__name__}",
    "backend": f"{fmt.backend.__module__}.{fmt.backend.__name__}",
    "configuration": options.model_dump(mode="json"),
    "effective_table_structure_options": {
        "type": type(options.table_structure_options).__name__,
        "mode": str(options.table_structure_options.mode),
        "do_cell_matching": options.table_structure_options.do_cell_matching,
    },
    "effective_heading_hierarchy_options": {
        "type": type(hh).__name__,
        "enabled": hh.enabled,
        "use_bookmarks": hh.use_bookmarks,
        "use_numbering": hh.use_numbering,
        "use_style": hh.use_style,
        "use_font_style": hh.use_font_style,
    },
    "generate_parsed_pages": options.generate_parsed_pages,
    "artifacts_path": str(ARTIFACTS),
    "pdf_dir": str(PDFS),
    "offline_env": {k: os.environ.get(k) for k in ("HF_HUB_OFFLINE", "TRANSFORMERS_OFFLINE")},
    "input_sha256": {k: hashlib.sha256(v.read_bytes()).hexdigest() for k, v in DOCS.items()},
    "documents": {},
    "memoria_metodo": f"campionamento continuo RSS su thread separato, intervallo {SAMPLE_INTERVAL_S}s",
}

t0 = time.perf_counter()
converter = DocumentConverter(format_options={InputFormat.PDF: fmt})
report["converter_init_seconds"] = round(time.perf_counter() - t0, 3)

for name, path in DOCS.items():
    t0 = time.perf_counter()
    result = converter.convert(path)
    elapsed = time.perf_counter() - t0
    doc = result.document
    md = doc.export_to_markdown()
    (OUT / f"candidate_{name}.md").write_text(md, encoding="utf-8", newline="\n")
    (OUT / f"candidate_{name}.json").write_text(
        json.dumps(doc.export_to_dict(), ensure_ascii=False, indent=2), encoding="utf-8", newline="\n"
    )
    intestazioni = []
    for it, _ in doc.iterate_items():
        if isinstance(it, TitleItem):
            intestazioni.append({"classe": "title", "livello": 0, "testo": it.text})
        elif isinstance(it, SectionHeaderItem):
            intestazioni.append({"classe": "section_header", "livello": it.level, "testo": it.text})
    report["documents"][name] = {
        "status": str(result.status),
        "pages": len(doc.pages),
        "seconds": round(elapsed, 3),
        "seconds_per_page": round(elapsed / len(doc.pages), 3),
        "tables": len(doc.tables),
        "table_shapes": [[t.data.num_rows, t.data.num_cols] for t in doc.tables],
        "intestazioni": intestazioni,
        "markdown_heading": [ln for ln in md.splitlines() if ln.lstrip().startswith("#")],
    }

_stop.set()
sampler.join(timeout=2)
report["memoria_mb"] = {
    "campioni": len(_samples),
    "picco": round(max(_samples) / 1e6, 1) if _samples else None,
    "mediana": round(sorted(_samples)[len(_samples) // 2] / 1e6, 1) if _samples else None,
}
(OUT / "run_report.json").write_text(
    json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8", newline="\n"
)
print(json.dumps({k: v for k, v in report.items() if k != "configuration"}, ensure_ascii=False, indent=2))

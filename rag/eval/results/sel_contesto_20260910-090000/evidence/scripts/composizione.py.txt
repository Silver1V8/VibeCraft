"""FASE 1/4 — composizione del contesto di /ask, IN-PROCESS e senza generazione.

Riusa il meccanismo di `eval/calibrate_graph_context.py`: stessa app, stesso
lifespan, stesso ranking, stesso ramo grafo, con il solo LLM sostituito da uno
stub. La generazione costa 10-60 s a chiamata e qui non serve: si misura **cosa
entra nel contesto**, non cosa il modello ne fa.

Per ogni domanda registra, per ogni blocco:
  · provenienza  semantico | grafo | codice
  · doc_id, sezione, classe dichiarata
  · caratteri, e la quota sul totale
e per ogni ESCLUSIONE del ramo grafo il motivo e il segnale usato.

Serve a rispondere, prima di qualunque batch di generazione:
  - la fonte attesa entra? da quale ramo?
  - quanto pesa ciascun ramo?
  - quali blocchi verrebbero esclusi da una regola di pertinenza, e perche'?

Uso (da rag/ del sandbox, server GIU', Qdrant su):
    composizione.py <out.json> [--top-k 10] [--set pdf|golden|multihop|holdout]
    Parametri sovrascrivibili: --graph-max, --graph-ratio, --code-total,
    --code-linked-max, --code-sem-max, --code-min-weight, --code-sem-min-score
"""
import argparse
import json
import pathlib
import re
import sys

import yaml

# RAG = la directory `rag/` DA CUI SI ESEGUE, cioe' quella del sandbox. Forzare qui il
# percorso del repository farebbe leggere a core.config il `database/` di produzione: e'
# successo, e la prima misura riportava l'index_version di produzione invece di quella del
# sandbox. Il cwd e' l'unica fonte corretta.
RAG = pathlib.Path.cwd()
if str(RAG) not in sys.path:
    sys.path.insert(0, str(RAG))

_BLOCCO_DOC = re.compile(r"^--- (?P<doc>.+?) § (?P<sec>.*?) \[(?P<meta>[^\]]*)\] ---$", re.M)
_BLOCCO_COD = re.compile(r"^--- Codice esistente: (?P<file>[^:]+?) :: (?P<sym>.+?) "
                         r"\((?P<kind>[^)]*)\)(?P<impl>.*?) ---$", re.M)
QUI = pathlib.Path(__file__).resolve().parent


def domande(quale: str) -> list[dict]:
    if quale == "pdf":
        y = yaml.safe_load((QUI.parent / "truth/domande_pdf_locali.yaml").read_text(encoding="utf-8"))
        return [{"id": d["id"], "q": d["domanda"], "attesa": d.get("risposta_attesa"),
                 "fonte": d["fonte"]} for d in y["domande"]]
    if quale == "holdout":
        y = yaml.safe_load((QUI.parent / "holdout/holdout.yaml").read_text(encoding="utf-8"))
        return [{"id": d["id"], "q": d["domanda"], "attesa": d.get("risposta_attesa"),
                 "fonte": d.get("fonte")} for d in y["domande"]]
    from eval.run_multihop import load_cases
    if quale == "multihop":
        return [{"id": c["q"][:40], "q": c["q"], "case": c}
                for c in load_cases() if c.get("kind") == "multihop"]
    y = yaml.safe_load((RAG / "eval/golden_set.yaml").read_text(encoding="utf-8"))
    casi = y["cases"] if isinstance(y, dict) else y
    return [{"id": c.get("id") or c["q"][:40], "q": c["q"], "case": c} for c in casi]


def blocchi(ctx: str) -> list[dict]:
    fuori = []
    for m in _BLOCCO_DOC.finditer(ctx or ""):
        meta = m.group("meta")
        fuori.append({"tipo": "grafo" if "via grafo" in meta else "semantico",
                      "doc": m.group("doc"), "sez": m.group("sec"), "meta": meta})
    for m in _BLOCCO_COD.finditer(ctx or ""):
        fuori.append({"tipo": "codice", "doc": m.group("file"), "sez": m.group("sym"),
                      "meta": m.group("kind")})
    return fuori


def quote(ctx: str) -> dict:
    """Caratteri per ramo. Il blocco codice e' introdotto da un'intestazione fissa."""
    MARK = "\n\n=== CODICE ESISTENTE (implementazione) ===\n\n"
    i = (ctx or "").find(MARK)
    doc, cod = ((ctx[:i], ctx[i:]) if i > 0 else (ctx or "", ""))
    pezzi = re.split(r"(?m)^(--- .*? ---)$", doc)
    sem = gra = 0
    for k in range(1, len(pezzi), 2):
        n = len(pezzi[k]) + len(pezzi[k + 1])
        if "via grafo" in pezzi[k]:
            gra += n
        else:
            sem += n
    return {"totale": len(ctx or ""), "semantico": sem, "grafo": gra, "codice": len(cod)}


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("uscita")
    ap.add_argument("--top-k", type=int, default=10)
    ap.add_argument("--set", default="pdf")
    ap.add_argument("--graph-max", type=int)
    ap.add_argument("--graph-ratio", type=float)
    ap.add_argument("--code-total", type=int)
    ap.add_argument("--code-linked-max", type=int)
    ap.add_argument("--code-sem-max", type=int)
    ap.add_argument("--code-min-weight", type=float)
    ap.add_argument("--code-sem-min-score", type=float)
    ap.add_argument("--code-link-score", type=float)
    a = ap.parse_args()

    from fastapi.testclient import TestClient

    from api import deps, routes_retrieval
    from core import llm
    from server import app

    deps.API_KEY = None

    async def _senza_llm(prompt: str, *, purpose: str) -> str:
        return ""
    llm.agenerate = _senza_llm
    routes_retrieval._cache_get = lambda app_, key: None
    routes_retrieval._cache_set = lambda app_, key, payload: None

    sovrascritti = {}
    for attr, val in (("GRAPH_CONTEXT_MAX_CHUNKS", a.graph_max),
                      ("GRAPH_CONTEXT_MIN_RATIO", a.graph_ratio),
                      ("CODE_CONTEXT_TOTAL_SNIPPETS", a.code_total),
                      ("CODE_CONTEXT_MAX_SNIPPETS", a.code_linked_max),
                      ("CODE_SEARCH_MAX_SNIPPETS", a.code_sem_max),
                      ("CODE_CONTEXT_MIN_WEIGHT", a.code_min_weight),
                      ("CODE_SEARCH_MIN_SCORE", a.code_sem_min_score),
                      ("CODE_LINK_MIN_QUESTION_SCORE", a.code_link_score)):
        if val is not None:
            setattr(routes_retrieval, attr, val)
            sovrascritti[attr] = val

    casi = domande(a.set)
    out = {"set": a.set, "top_k": a.top_k, "parametri_sovrascritti": sovrascritti, "casi": {}}
    with TestClient(app, client=("127.0.0.1", 54321)) as client:
        attuali = {n: getattr(routes_retrieval, n) for n in
                   ("GRAPH_CONTEXT_MAX_CHUNKS", "GRAPH_CONTEXT_MIN_RATIO",
                    "CODE_CONTEXT_TOTAL_SNIPPETS", "CODE_CONTEXT_MAX_SNIPPETS",
                    "CODE_SEARCH_MAX_SNIPPETS", "CODE_CONTEXT_MIN_WEIGHT",
                    "CODE_SEARCH_MIN_SCORE", "CODE_LINK_MIN_QUESTION_SCORE")}
        out["parametri_effettivi"] = attuali
        for c in casi:
            r = client.post("/ask", json={"question": c["q"], "top_k": a.top_k})
            r.raise_for_status()
            j = r.json()
            ctx = (j.get("graph_context") or {}).get("context_text")
            if ctx is None:
                out["casi"][c["id"]] = {"router": "deterministico", "domanda": c["q"]}
                continue
            bl = blocchi(ctx)
            out["casi"][c["id"]] = {
                "domanda": c["q"], "fonte_attesa": c.get("fonte"),
                "quote": quote(ctx), "blocchi": bl,
                "n_semantici": sum(b["tipo"] == "semantico" for b in bl),
                "n_grafo": sum(b["tipo"] == "grafo" for b in bl),
                "n_codice": sum(b["tipo"] == "codice" for b in bl),
                "contesto": ctx,
            }
    pathlib.Path(a.uscita).write_text(json.dumps(out, indent=2, ensure_ascii=False), encoding="utf-8")
    vivi = [v for v in out["casi"].values() if "quote" in v]
    if vivi:
        med = lambda k: sorted(v["quote"][k] for v in vivi)[len(vivi) // 2]  # noqa: E731
        print(f"  casi {len(vivi)} | mediana caratteri: totale {med('totale')} "
              f"(sem {med('semantico')}, grafo {med('grafo')}, codice {med('codice')})")
        print(f"  mediana blocchi: sem {sorted(v['n_semantici'] for v in vivi)[len(vivi)//2]}, "
              f"grafo {sorted(v['n_grafo'] for v in vivi)[len(vivi)//2]}, "
              f"codice {sorted(v['n_codice'] for v in vivi)[len(vivi)//2]}")
    print(f"-> {a.uscita}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

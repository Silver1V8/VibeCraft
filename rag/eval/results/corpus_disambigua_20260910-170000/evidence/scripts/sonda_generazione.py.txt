"""Prova mirata ed economica: genera le risposte sui contesti gia' costruiti dalla
sonda di composizione, con il prompt ORIGINALE congelato.

Serve a decidere se una variante di selezione merita il batch fattoriale completo,
senza pagarlo. I contesti sono quelli veri, prodotti in-process dal percorso
applicativo; qui si aggiunge soltanto la generazione.

NON sostituisce la verifica finale, che deve attraversare il percorso completo
via HTTP: e' lo screening che la precede.

Uso: sonda_generazione.py <composizione.json> <out.json> <passate> [id...]
"""
import json
import pathlib
import re
import sys
import time

RAG = pathlib.Path(r"C:\Users\Silve\Desktop\vibecraft\rag")
sys.path.insert(0, str(RAG))

from core import llm, retrieval  # noqa: E402

INT = re.compile(r"^--- (?P<doc>\S+) § (?P<sez>.*?) \[(?P<cls>[^\]]+)\] ---$", re.M)
PESI = {"IRON_MEMORY": 1.0, "AUTHORIZATION_POLICY": 0.9, "NORMS": 0.9, "REQUIREMENTS": 0.87,
        "DATA_DICTIONARY": 0.7, "METADATA": 0.6, "METHODOLOGY": 0.5, "SENZA CLASSE": 0.3}
DINIEGO = re.compile(r"non\s+(?:è|e')?\s*(?:riport\w+|contien\w+|specific\w+|indic\w+|"
                     r"present\w+|fornis\w+)|nessun\w*\s+valore", re.I)
COEFF = re.compile(r"\b0[.,](?:30|50|60|70|85|87|90|95)\b")


def scala(ctx: str):
    viste = {}
    for m in INT.finditer(ctx):
        c = m.group("cls").split(";")[0].split(",")[0].strip().upper()
        viste.setdefault(c, PESI.get(c, 0.3))
    return sorted(viste.items(), key=lambda kv: -kv[1]) if len(viste) >= 2 else []


def main() -> int:
    sorgente, uscita, passate = sys.argv[1], sys.argv[2], int(sys.argv[3])
    solo = set(sys.argv[4:]) or None
    dati = json.loads(pathlib.Path(sorgente).read_text(encoding="utf-8"))
    out = {"sorgente": sorgente, "parametri": dati.get("parametri_effettivi"),
           "modello": llm.resolve("ask"), "temperature": llm.OLLAMA_TEMPERATURE,
           "prompt": "ORIGINALE (congelato)", "casi": {}}
    for qid, c in dati["casi"].items():
        if solo and qid not in solo:
            continue
        if "contesto" not in c:
            continue
        ctx = c["contesto"]
        prompt = retrieval.build_ask_prompt(ctx, c["domanda"], authority_classes=scala(ctx))
        righe = []
        for i in range(passate):
            t0 = time.perf_counter()
            try:
                r, err = llm.generate(prompt, purpose="ask"), None
            except Exception as exc:  # noqa: BLE001
                r, err = "", f"{type(exc).__name__}: {exc}"
            righe.append({"passata": i + 1, "risposta": r, "errore": err,
                          "secondi": round(time.perf_counter() - t0, 2),
                          "nega": bool(DINIEGO.search(r)),
                          "coefficiente": bool(COEFF.search(r))})
        out["casi"][qid] = {"domanda": c["domanda"], "prompt_char": len(prompt),
                            "prompt_token_stimati": len(prompt) // 3,
                            "quote": c.get("quote"), "passate": righe}
        print(f"  {qid:14} prompt {len(prompt)//3:6} tok | " +
              " | ".join(f"p{r['passata']} {r['secondi']:5.1f}s" for r in righe))
    pathlib.Path(uscita).write_text(json.dumps(out, indent=2, ensure_ascii=False), encoding="utf-8")
    print(f"-> {uscita}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

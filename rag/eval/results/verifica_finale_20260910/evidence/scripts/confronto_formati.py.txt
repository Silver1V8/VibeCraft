def _hyperlink(paragraph, url, text):
    from docx.opc.constants import RELATIONSHIP_TYPE as RT
    from docx.oxml import OxmlElement
    from docx.oxml.ns import qn
    r_id = paragraph.part.relate_to(url, RT.HYPERLINK, is_external=True)
    link = OxmlElement("w:hyperlink")
    link.set(qn("r:id"), r_id)
    run = OxmlElement("w:r")
    t = OxmlElement("w:t")
    t.text = text
    run.append(t)
    link.append(run)
    paragraph._p.append(link)

def heads(md):
    return [(len(m.group(1)), m.group(2).strip()) for m in re.finditer(r"(?m)^(#{1,6})\s+(.*)$", md)]


def lvl(md, text):
    return next((n for n, t in heads(md) if text in t), None)


def riga(md, *cells):
    return any(ln.lstrip().startswith("|") and all(c in ln for c in cells) for ln in md.splitlines())


def voce(md, text):
    return re.search(rf"(?m)^\s*([-*+]|\d+[.)])\s+.*{re.escape(text)}", md) is not None


def rientro(md, text):
    ln = next((ln for ln in md.splitlines() if text in ln), None)
    return None if ln is None else len(ln) - len(ln.lstrip())


def prima(md, a, b):
    return 0 <= md.find(a) < md.find(b)


def tutti(md, *parole):
    return all(p in md for p in parole)


def gerarchia(md, *titoli):
    livelli = [lvl(md, t) for t in titoli]
    return None not in livelli and all(a < b for a, b in zip(livelli, livelli[1:]))

_generic = None


def _docling_generic():
    global _generic
    if _generic is None:
        from docling.datamodel.base_models import InputFormat
        from docling.document_converter import DocumentConverter
        _generic = DocumentConverter(allowed_formats=[
            InputFormat.DOCX, InputFormat.PPTX, InputFormat.XLSX, InputFormat.XLS, InputFormat.HTML,
            InputFormat.CSV, InputFormat.MD, InputFormat.EPUB, InputFormat.EMAIL])
    return _generic


def _finisci(md: str, nome: str) -> str:
    md, _n = ic._LOCAL_IMAGE_RE.subn("", md)
    md = normalize_markdown(md)
    if md and not ic._HEADING_RE.search(md):
        md = f"# {Path(nome).stem}\n\n{md}"
    return md


def docling(data: bytes, nome: str, fogli: bool = False) -> str:
    from docling.datamodel.base_models import DocumentStream
    r = _docling_generic().convert(DocumentStream(name=nome, stream=io.BytesIO(data)), raises_on_error=False)
    if r.status.value not in ("success", "partial_success"):
        raise RuntimeError(f"docling {r.status.value}: {[str(e.error_message) for e in r.errors]}")
    d = r.document
    if not fogli:
        return _finisci(d.export_to_markdown(escape_underscores=False, image_placeholder=""), nome)
    parti = []
    for grp in d.body.children:
        g = grp.resolve(d)
        nome_foglio = getattr(g, "name", "") or ""
        testo = d.export_to_markdown(escape_underscores=False, image_placeholder="",
                                     page_no=(g.prov[0].page_no if getattr(g, "prov", None) else None))
        parti.append(f"## {nome_foglio}\n\n{testo}")
    return _finisci("\n\n".join(parti), nome)


def markitdown(data: bytes, fmt: str, nome: str) -> str:
    ic.INTAKE_DOCLING_ENABLED = False
    try:
        return ic.convert_to_markdown(data, fmt, filename=nome).markdown
    finally:
        ic.INTAKE_DOCLING_ENABLED = True


def docling_pdf(data: bytes, nome: str) -> tuple[str, dict]:
    ic.INTAKE_DOCLING_ENABLED = True
    r = ic.convert_to_markdown(data, "pdf", filename=nome)
    return r.markdown, r.metadata


def misura(fn):
    proc = psutil.Process()
    rss0, t0 = proc.memory_info().rss, time.perf_counter()
    try:
        md, err = fn(), None
    except Exception as exc:  # l'errore è un esito del confronto
        md, err = "", f"{type(exc).__name__}: {exc}"[:300]
    return md, err, round(time.perf_counter() - t0, 3), round((proc.memory_info().rss - rss0) / 1e6, 1)


def main() -> int:
    pdf = fixture_pdfs()
    casi = {
        "docx": ("docx", "manuale.docx", fixture_docx()),
        "pptx": ("pptx", "rilascio.pptx", fixture_pptx()),
        "xlsx": ("xlsx", "budget.xlsx", fixture_xlsx()),
        "html": ("html", "guida.html", HTML.format(cs="utf-8").encode("utf-8")),
        "html_latin1": ("html", "guida_latin1.html", HTML.format(cs="iso-8859-1").replace("≥", "&ge;").encode("latin-1")),
        "csv": ("csv", "listino.csv", CSV_UTF8.encode("utf-8")),
        "csv_semicolon": ("csv", "citta.csv", CSV_SEMI.encode("latin-1")),
        "txt": ("txt", "note.txt", TXT.encode("utf-8")),
        "json": ("json", "progetto.json", JSON.encode("utf-8")),
        "xml": ("xml", "catalogo.xml", XML.encode("utf-8")),
        "epub": ("epub", "libro.epub", fixture_epub()),
        "pdf_digitale": ("pdf", "registro.pdf", pdf["digitale"]),
        "pdf_scansione": ("pdf", "scansione.pdf", pdf["scansione"]),
        "pdf_misto": ("pdf", "misto.pdf", pdf["misto"]),
    }
    if XLS and XLS.is_file():
        casi["xls"] = ("xls", "budget.xls", XLS.read_bytes())
    report: dict = {"casi": {}, "errori_attesi": {}}
    for caso, (fmt, nome, data) in casi.items():
        (FIX / nome).write_bytes(data)
        motori = {"markitdown": lambda: markitdown(data, fmt, nome)}
        if fmt == "pdf":
            motori["docling"] = lambda: docling_pdf(data, nome)[0]
        elif fmt in ("docx", "pptx", "xlsx", "xls", "html", "csv", "txt", "epub"):
            motori["docling"] = lambda: docling(data, nome)
        if caso == "html_latin1":
            motori["docling+utf8"] = lambda: docling(data.decode(ic._detect_charset(data)).encode("utf-8"), nome)
        if fmt in ("xlsx", "xls"):
            motori["docling+fogli"] = lambda: docling(data, nome, fogli=True)
        esiti = {}
        for motore, fn in motori.items():
            md, err, sec, mb = misura(fn)
            (OUT / f"{caso}.{motore}.md").write_text(md, encoding="utf-8")
            controlli = {nome_c: bool(md) and bool(c(md)) for nome_c, c in ATTESE.get(caso, [])}
            esiti[motore] = {"errore": err, "secondi": sec, "rss_delta_mb": mb, "caratteri": len(md),
                             "sezioni_chunk": len(chunk_markdown("x.md", {"id": "x"}, md)) if md else 0,
                             "titoli": heads(md)[:12], "controlli": controlli,
                             "superati": f"{sum(controlli.values())}/{len(controlli)}"}
        report["casi"][caso] = esiti
        print(f"{caso:14} " + " | ".join(f"{m}: {e['superati']} {e['secondi']}s" + (f" ERR {e['errore'][:60]}" if e["errore"] else "")
                                         for m, e in esiti.items()), flush=True)
    # File problematici: il percorso applicativo deve dare un errore chiaro, non un documento.
    for caso, (fmt, nome, data) in {"pdf_protetto": ("pdf", "protetto.pdf", pdf["protetto"]),
                                     "docx_corrotto": ("docx", "rotto.docx", b"PK\x03\x04 non uno zip"),
                                     "xlsx_vuoto": ("xlsx", "vuoto.xlsx", b""),
                                     "pptx_corrotto": ("pptx", "rotto.pptx", b"non un pptx")}.items():
        try:
            r = ic.convert_to_markdown(data, fmt, filename=nome)
            report["errori_attesi"][caso] = {"esito": "CONVERTITO", "motore": r.metadata["motore"], "avvisi": r.metadata["avvisi"]}
        except ic.ConversionError as exc:
            report["errori_attesi"][caso] = {"esito": "ConversionError", "messaggio": str(exc)[:200]}
        except Exception as exc:  # un'eccezione grezza è un difetto del percorso
            report["errori_attesi"][caso] = {"esito": f"ECCEZIONE GREZZA {type(exc).__name__}", "messaggio": str(exc)[:200]}
        print(caso, report["errori_attesi"][caso], flush=True)
    report["tempi_mediani"] = {m: statistics.median(e[m]["secondi"] for e in report["casi"].values() if m in e)
                               for m in ("markitdown", "docling")}
    (OUT / "confronto.json").write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
    return 0

"""Attese del banco Office/LibreOffice, scritte PRIMA di eseguire i confronti.

I valori memorizzati delle formule vengono dalla lettura delle fixture con lettori
indipendenti dai convertitori (openpyxl data_only per XLSX, xlrd per XLS), fatta prima
delle misure:
  registro.xlsx        (openpyxl)  C7=999 (obsoleto: il ricalcolo darebbe 10), E7=1500.5, F7 assente
  registro_excel.xlsx  (Excel 16)  C7=10, E7=1500.5, F7=2401
  registro.xls         (Excel 16)  C7=10, E7=1500.5, F7=2401
Un valore diverso da quello memorizzato è un CAMBIAMENTO, non un miglioramento.
"""
import re

MEMORIZZATI = {
    "registro.xlsx": {"C7": "999", "E7": "1500.5", "F7": None},
    "registro_excel.xlsx": {"C7": "10", "E7": "1500.5", "F7": "2401"},
    "registro.xls": {"C7": "10", "E7": "1500.5", "F7": "2401"},
}
# Valori che un ricalcolo produrrebbe per le formule di registro.xlsx.
RICALCOLATI = {"C7": "10", "F7": "2401"}


def heads(md):
    return [(len(m.group(1)), m.group(2).strip()) for m in re.finditer(r"(?m)^(#{1,6})\s+(.*)$", md)]


def lvl(md, text):
    return next((n for n, t in heads(md) if text in t), None)


def celle(riga: str) -> list[str]:
    return [c.strip() for c in riga.strip().strip("|").split("|")]


def righe_tabella(md):
    return [celle(ln) for ln in md.splitlines()
            if ln.lstrip().startswith("|") and not set(ln.strip()) <= set("|-: ")]


def riga(md, *valori):
    """Una riga di tabella con TUTTI i valori, ciascuno uguale a una cella: l'associazione
    fra etichetta e valore è verificata dalla riga, non dalla sola presenza del testo."""
    return any(all(v in r for v in valori) for r in righe_tabella(md))


def voce(md, text):
    return re.search(rf"(?m)^\s*([-*+]|\d+[.)])\s+.*{re.escape(text)}", md) is not None


def rientro(md, text):
    ln = next((ln for ln in md.splitlines() if text in ln), None)
    return None if ln is None else len(ln) - len(ln.lstrip())


def prima(md, a, b):
    return 0 <= md.find(a) < md.find(b)


def tutti(md, *parole):
    return all(p in md for p in parole)


def una_volta(md, *parole):
    return all(md.count(p) == 1 for p in parole)


def gerarchia(md, *titoli):
    livelli = [lvl(md, t) for t in titoli]
    return None not in livelli and all(a < b for a, b in zip(livelli, livelli[1:]))


def _num(c):
    try:
        return float(c.replace(" ", ""))
    except ValueError:
        return None


def riga_totale(md):
    return next((r for r in righe_tabella(md) if r and r[0] == "Totale"), None)


def formule_memorizzate(md, nome):
    """Ogni valore memorizzato compare nella riga Totale; una formula SENZA valore memorizzato
    non riceve un numero inventato (né il ricalcolo)."""
    r = riga_totale(md)
    if r is None:
        return False
    numeri = {_num(c) for c in r} - {None}
    attesi = MEMORIZZATI[nome]
    for ref, v in attesi.items():
        if v is not None and float(v) not in numeri:
            return False
        if v is None and float(RICALCOLATI[ref]) in numeri:
            return False
    return True


def osservato_c7(md):
    r = riga_totale(md)
    if r is None:
        return "riga Totale assente"
    numeri = {_num(c) for c in r} - {None}
    return "999 (memorizzato)" if 999.0 in numeri else "10 (ricalcolato)" if 10.0 in numeri else "assente"

def problemi_markdown(md: str) -> list[str]:
    """Validità: tabelle con lo stesso numero di colonne in ogni riga e la riga separatrice;
    nessun tag HTML (i commenti sono ammessi); righe di testo duplicate."""
    out, blocco = [], []
    for ln in md.splitlines() + [""]:
        if ln.lstrip().startswith("|"):
            blocco.append(ln)
            continue
        if blocco:
            n = {ln2.strip().count("|") for ln2 in blocco}
            if len(n) > 1:
                out.append(f"tabella con colonne diverse: {sorted(n)}")
            if len(blocco) > 1 and not set(blocco[1].strip()) <= set("|-: "):
                out.append("tabella senza riga separatrice")
            blocco = []
    if re.search(r"<(?!!--)[a-zA-Z/][^>]*>", md):
        out.append("tag HTML")
    testo = [ln.strip() for ln in md.splitlines() if ln.strip() and not ln.lstrip().startswith(("|", "#"))]
    dup = {t for t in testo if testo.count(t) > 1 and len(t) > 12}
    if dup:
        out.append(f"righe duplicate: {sorted(dup)[:3]}")
    return out

"""FASE 2 — l'effetto e' monotono nella DIMENSIONE del contesto?

Prompt ORIGINALE, invariato. Si tiene il blocco del PDF e si aggiungono i primi
k blocchi concorrenti nell'ordine di rango del retrieval. Se il tasso di successo
decade al crescere di k, la causa non e' la formulazione del prompt: e' la
diluizione del contesto, e nessuna frase aggiuntiva la corregge.
"""
import json
import pathlib
import sys
import re
RAG = pathlib.Path(r"C:\Users\Silve\Desktop\vibecraft\rag")
sys.path.insert(0, str(RAG))
from core import llm, retrieval  # noqa: E402
ARCH = RAG / "eval/results/fix_attribuzione_20260909-183000/out"
cfg, n, uscita = sys.argv[1], int(sys.argv[2]), sys.argv[3]
d = json.loads((ARCH/f"eval_{cfg}_p1.json").read_text(encoding="utf-8"))
q = d["domande"]["pdf-diag-3"]
ctx = q["contesto_inviato"]
pezzi = re.split(r"(?m)^(--- .*? ---)$", ctx)
blocchi = [(pezzi[i], pezzi[i+1]) for i in range(1, len(pezzi), 2)]
pdf = [b for b in blocchi if "catalogo_diagrammi" in b[0]]
altri = [b for b in blocchi if "catalogo_diagrammi" not in b[0]]
scala=[("NORMS",0.9),("REQUIREMENTS",0.87),("METADATA",0.6),("METHODOLOGY",0.5),("SENZA CLASSE",0.3)]
res={}
for k in (0, 2, 4, 8, len(altri)):
    c = "".join(h+t for h,t in pdf + altri[:k])
    righe=[]
    for i in range(n):
        p = retrieval.build_ask_prompt(c, q["domanda"], authority_classes=scala)
        r = llm.generate(p, purpose="ask")
        kb="19.5 KB" in r
        coeff=bool(re.search(r"\b0[.,](?:30|50|60|85|87|90)\b", r))
        nega=bool(re.search(r"non\s+(?:è|e')?\s*(?:riport\w+|contien\w+|specific\w+|indic\w+|present\w+)|nessun\w*\s+valore", r, re.I))
        righe.append({"r":i+1,"kb":kb,"coeff":coeff,"nega":nega,"risposta":r})
    pulite=sum(x["kb"] and not x["coeff"] and not x["nega"] for x in righe)
    tok=len(p)//3
    print(f"  PDF + {k:2} blocchi | prompt {tok:6} tok | 19.5KB {sum(x['kb'] for x in righe)}/{n} | coeff {sum(x['coeff'] for x in righe)}/{n} | PULITE {pulite}/{n}")
    res[f"pdf+{k}"]={"token_prompt":tok,"contesto_char":len(c),"righe":righe}
pathlib.Path(uscita).write_text(json.dumps(res,indent=2,ensure_ascii=False),encoding="utf-8")

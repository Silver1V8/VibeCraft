"""FASE 2 — ablazione: quale ingrediente del percorso condiviso produce D1?

Quattro condizioni sul contesto Docling archiviato, prompt ORIGINALE invariato:
  1 completo + scala        (controllo: deve fallire)
  2 senza il blocco README  (il README porta una tabella documento->peso numerico)
  3 completo, senza scala   ([CLASSE-FONTI] assente: niente numeri nel prompt)
  4 senza README, senza scala
Se 2 corregge, la causa e' il contesto
se 3 corregge, e' il prompt
se serve 4,
sono concorrenti. Nessuna modifica al prompt: si ablaziona, non si riscrive.
"""
import json
import pathlib
import sys
import re
RAG = pathlib.Path(r"C:\Users\Silve\Desktop\vibecraft\rag")
sys.path.insert(0, str(RAG))
from core import llm, retrieval  # noqa: E402
ARCH = RAG / "eval/results/fix_attribuzione_20260909-183000/out"
cfg, n, uscita = sys.argv[1], int(sys.argv[2]), sys.argv[3]
d = json.loads((ARCH/f"eval_{cfg}_p1.json").read_text(encoding="utf-8"))
q = d["domande"]["pdf-diag-3"]
ctx = q["contesto_inviato"]
INT = re.compile(r"^--- (?P<doc>\S+) § (?P<sez>.*?) \[(?P<cls>[^\]]+)\] ---$", re.M)
PESI = {"NORMS":0.9,"REQUIREMENTS":0.87,"METADATA":0.6,"METHODOLOGY":0.5,"SENZA CLASSE":0.3}
viste = {}
for m in INT.finditer(ctx):
    c = m.group("cls").split(";")[0].split(",")[0].strip().upper()
    viste.setdefault(c, PESI.get(c,0.3))
scala = sorted(viste.items(), key=lambda kv:-kv[1])

def senza_readme(c):
    pezzi = re.split(r"(?m)^(--- .*? ---)$", c)
    fuori=[pezzi[0]]
    for i in range(1,len(pezzi),2):
        if pezzi[i].startswith("--- README.md "):
            continue
        fuori += [pezzi[i], pezzi[i+1]]
    return "".join(fuori)

COND = [("1 completo+scala", ctx, scala), ("2 senzaREADME+scala", senza_readme(ctx), scala),
        ("3 completo-scala", ctx, None), ("4 senzaREADME-scala", senza_readme(ctx), None)]
res={}
for nome, c, s in COND:
    righe=[]
    for i in range(n):
        p = retrieval.build_ask_prompt(c, q["domanda"], authority_classes=s)
        r = llm.generate(p, purpose="ask")
        kb = "19.5 KB" in r
        auth = bool(re.search(r"\b0[.,](?:30|50|60|85|87|90)\b", r))
        nega = bool(re.search(r"non\s+(?:è|e')?\s*(?:riport\w+|contien\w+|specific\w+|indic\w+|present\w+)|nessun\w*\s+valore", r, re.I))
        righe.append({"r":i+1,"kb":kb,"coeff":auth,"nega":nega,"risposta":r})
    ok=sum(x["kb"] and not x["coeff"] and not x["nega"] for x in righe)
    print(f"  {nome:22} 19.5KB {sum(x['kb'] for x in righe)}/{n} | coeff {sum(x['coeff'] for x in righe)}/{n} | nega {sum(x['nega'] for x in righe)}/{n} | PULITE {ok}/{n}")
    res[nome]={"prompt_char":len(p),"righe":righe}
pathlib.Path(uscita).write_text(json.dumps(res, indent=2, ensure_ascii=False), encoding="utf-8")

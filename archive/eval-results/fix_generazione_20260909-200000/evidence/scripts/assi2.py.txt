`classifica_riferimento()` separa, con un controllo eseguibile per ciascuna:

  documento     percorso o nome di file

  descrizione   prosa che non e' nessuna delle precedenti

# Etichette STRUTTURALI del contesto: non sono contenuto di una sezione.
# Le classi note vengono dalla configurazione applicativa; le altre si
# raccolgono dal contesto realmente inviato (vedi `etichette_dal_contesto`).
STATI_ARCHIVIO = ("ARCHIVED", "DEPRECATED", "HISTORICAL", "OBSOLETE", "SUPERSEDED")
ETICHETTE_FISSE = {
    "SENZA CLASSE", "CODICE ESISTENTE", "IRON_MEMORY", "AUTHORIZATION_POLICY",
    "DATA_DICTIONARY", "METHODOLOGY", "NORMS", "REQUIREMENTS", "METADATA",
    "KNOWLEDGE", "SPECS", *STATI_ARCHIVIO,
}
_ESTENSIONI = r"\.(?:pdf|md|html|svg|json|yaml|yml|txt|py|png)$"
_INTESTAZIONE = re.compile(r"^---\s+(?P<doc>\S+)\s+§\s*(?P<sez>.*?)\s*\[(?P<cls>[^\]]+)\]\s+---$",
                           re.MULTILINE)


def etichette_dal_contesto(contesto: str) -> set[str]:
    """Le etichette di classe REALMENTE presenti nelle intestazioni di blocco del
    contesto inviato. Prese dal dato, non indovinate: se il corpus dichiara una
    classe nuova, lo schermo la riconosce senza dover essere aggiornato."""
    trovate = set()
    for m in _INTESTAZIONE.finditer(contesto or ""):
        for pezzo in m.group("cls").split(","):
            if pezzo.strip():
                trovate.add(pezzo.strip().upper())
    return trovate


def e_etichetta(valore: str, etichette: set[str] | None = None) -> bool:
    v = re.sub(r"^[\[\s]+|[\]\s]+$", "", valore or "").strip().upper()
    if not v:
        return False
    note = ETICHETTE_FISSE | (etichette or set())
    if v in note:
        return True
    # «NORMS, ARCHIVED» o «SENZA CLASSE, DEPRECATED»
    pezzi = [p.strip() for p in v.split(",") if p.strip()]
    return bool(pezzi) and all(p in note for p in pezzi)


def e_documento(valore: str) -> bool:
    return bool(re.search(_ESTENSIONI, (valore or "").strip(), re.IGNORECASE))


def classifica_riferimento(valore: str, fonte: str = "", etichette: set[str] | None = None) -> str:
    """classe | documento | sezione | citazione | descrizione."""
    v = (valore or "").strip()
    if not v:
        return "descrizione"
    if e_etichetta(v, etichette):
        return "classe"
    if e_documento(v):
        return "documento"
    titoli = {normalize(t) for t, _ in sezioni_del_testo(fonte)}
    if normalize(v) in titoli:
        return "sezione"
    if normalize(v) and normalize(v) in normalize(fonte):
        return "citazione"
    return "descrizione"


ESITI_ERRORE = ("sezione_inesistente", "valore_in_altra_sezione",
                "valore_non_trovato", "fonte_senza_struttura")


def errori_di_sezione(esiti: list[dict]) -> list[dict]:
    return [e for e in esiti if e.get("esito") in ESITI_ERRORE]


def controlla_sezioni(risposta: str, testo_fonte: str,
                      contesto: str = "") -> list[dict]:
    """Come `assi.controlla_sezioni`, ma i «valori» attribuiti a una sezione
    escludono etichette di classe, percorsi di documento e il titolo stesso."""
    sezioni = sezioni_del_testo(testo_fonte)
    titoli = {normalize(t): (t, c) for t, c in sezioni}
    etichette = etichette_dal_contesto(contesto)
    esiti: list[dict] = []
    for m in SEZIONE.finditer(risposta or ""):
        nome = m.group(1).strip()
        inizio = risposta.rfind("\n", 0, m.start()) + 1
        fine = risposta.find("\n", m.end())
        frase = risposta[inizio: fine if fine > 0 else len(risposta)]
        grezzi = re.findall(r"[\"«“]([^\"»”\n]{3,120})[\"»”]", frase)
        grezzi += re.findall(r"`([^`\n]{3,120})`", frase)

        valori, scartati = [], []
        for v in grezzi:
            if not normalize(v) or normalize(v) == normalize(nome):
                scartati.append({"valore": v[:80], "tipo": "sezione"})
                continue
            tipo = classifica_riferimento(v, testo_fonte, etichette)
            if tipo in ("classe", "documento"):
                scartati.append({"valore": v[:80], "tipo": tipo})
                continue
            valori.append(v)

        if not sezioni:
            esiti.append({"sezione_dichiarata": nome, "esito": "fonte_senza_struttura",
                          "nota": "il testo indicizzato non ha intestazioni: nessuna "
                                  "affermazione di sezione e' sostenibile da questa fonte"})
            continue
        chiave = normalize(nome)
        trovata = next((k for k in titoli if k == chiave), None)
        if trovata is None:
            trovata = next((k for k in titoli if chiave and (chiave in k or k in chiave)), None)
        if trovata is None:
            esiti.append({"sezione_dichiarata": nome, "esito": "sezione_inesistente",
                          "intestazioni_reali": [t for t, _ in sezioni],
                          "scartati_non_valori": scartati})
            continue
        titolo_reale, corpo = titoli[trovata]
        for v in valori:
            nv = normalize(v)
            if nv in normalize(corpo):
                esiti.append({"sezione_dichiarata": nome, "sezione_reale": titolo_reale,
                              "valore": v[:80], "esito": "coerente"})
                continue
            altrove = [t for t, c in sezioni if nv in normalize(c)]
            esiti.append({
                "sezione_dichiarata": nome, "sezione_reale": titolo_reale, "valore": v[:80],
                "esito": "valore_in_altra_sezione" if altrove else "valore_non_trovato",
                "sezione_effettiva": altrove[:2],
            })
        if not valori:
            esiti.append({"sezione_dichiarata": nome, "sezione_reale": titolo_reale,
                          "esito": "sezione_corretta_senza_valore",
                          "scartati_non_valori": scartati})
    return esiti

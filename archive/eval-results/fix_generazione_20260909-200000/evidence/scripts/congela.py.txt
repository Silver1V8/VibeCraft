def sha(p: pathlib.Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def corpus_hash(root: pathlib.Path) -> dict:
    """Inventario del corpus: ogni file con dimensione e SHA-256. I PDF
    RIGENERATI devono avere hash DIVERSI dagli originali di produzione, ed e'
    la differenza che rende il perimetro dichiarabile."""
    voci = {}
    for p in sorted(root.rglob("*")):
        if p.is_file():
            voci[str(p.relative_to(root)).replace("\\", "/")] = {
                "byte": p.stat().st_size, "sha256": sha(p)}
    return voci


def collections(url: str) -> list[str]:
    try:
        with urllib.request.urlopen(url + "/collections", timeout=30) as r:
            d = json.loads(r.read().decode("utf-8"))
        return sorted(c["name"] for c in d["result"]["collections"])
    except Exception as exc:  # noqa: BLE001
        return [f"ERRORE: {type(exc).__name__}: {exc}"]

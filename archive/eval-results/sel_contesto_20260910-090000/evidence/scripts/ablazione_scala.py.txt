"""DIAGNOSTICO (non una proposta): a contesto ridotto, la scala [CLASSE-FONTI] e' cio' che resta?

`authority_classes` viene da `authority_scale(fonti_nel_contesto)`: la scala compare
perche' il contesto mescola classi diverse. Qui la si toglie a mano, a parita' di
contesto, per isolare il suo contributo residuo. NON e' una modifica proposta: il
prompt di questa sessione resta l'originale congelato.
"""
import json
import pathlib
import re
import sys
RAG = pathlib.Path(r"C:\Users\Silve\Desktop\vibecraft\rag")
sys.path.insert(0, str(RAG))
from core import llm, retrieval  # noqa: E402
INT = re.compile(r"^--- (?P<doc>\S+) § (?P<sez>.*?) \[(?P<cls>[^\]]+)\] ---$", re.M)
PESI = {"NORMS":0.9,"REQUIREMENTS":0.87,"METADATA":0.6,"METHODOLOGY":0.5,"SENZA CLASSE":0.3}
COEFF = re.compile(r"\b0[.,](?:30|50|60|70|85|87|90|95)\b")
NEGA  = re.compile(r"non\s+(?:è|e')?\s*(?:riport\w+|contien\w+|specific\w+|indic\w+|present\w+|fornis\w+)|nessun\w*\s+valore", re.I)
sorgente, uscita, n = sys.argv[1], sys.argv[2], int(sys.argv[3])
d = json.loads(pathlib.Path(sorgente).read_text(encoding="utf-8"))
c = d["casi"]["pdf-diag-3"]
ctx = c["contesto"]
viste = {}
for m in INT.finditer(ctx):
    k = m.group("cls").split(";")[0].split(",")[0].strip().upper()
    viste.setdefault(k, PESI.get(k, 0.3))
scala = sorted(viste.items(), key=lambda kv: -kv[1]) if len(viste) >= 2 else []
res = {}
for nome, sc in (("con scala", scala), ("SENZA scala", None)):
    p = retrieval.build_ask_prompt(ctx, c["domanda"], authority_classes=sc)
    righe = []
    for i in range(n):
        r = llm.generate(p, purpose="ask")
        kb = "19.5 KB" in r
        righe.append({"passata": i+1, "kb": kb, "coeff": bool(COEFF.search(r)),
                      "nega": bool(NEGA.search(r)), "risposta": r})
    pulite = sum(x["kb"] and not x["coeff"] and not x["nega"] for x in righe)
    print(f"  {nome:12} ({len(p)//3:5} tok): 19.5KB {sum(x['kb'] for x in righe)}/{n} | "
          f"coeff {sum(x['coeff'] for x in righe)}/{n} | PULITE {pulite}/{n}")
    res[nome] = {"token": len(p)//3, "righe": righe}
pathlib.Path(uscita).write_text(json.dumps(res, indent=2, ensure_ascii=False), encoding="utf-8")

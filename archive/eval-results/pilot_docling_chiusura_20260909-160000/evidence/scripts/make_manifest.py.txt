def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def versioni(python):
    out = subprocess.run([python, "-m", "pip", "freeze"], capture_output=True, text=True)
    d = {}
    for r in out.stdout.splitlines():
        if "==" in r:
            k, v = r.split("==", 1)
            d[k.lower()] = v
    return d

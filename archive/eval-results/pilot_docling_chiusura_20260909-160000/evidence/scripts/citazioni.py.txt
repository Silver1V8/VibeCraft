# estrattore DIFETTOSO, conservato solo per il confronto prima/dopo
REGEX_DIFETTOSA = re.compile(r"[«\"']([^«»\"']{12,200})[»\"']")


def estrai_difettoso(risposta: str) -> list[str]:
    return [" ".join(m.split()) for m in REGEX_DIFETTOSA.findall(risposta)]


def estrai(risposta: str) -> list[str]:
    """I passi che la risposta presenta come parole di una fonte."""
    return quoted_spans(risposta)

    INDICIZZATO, cioe' esattamente cio' che e' finito nel contesto."""
